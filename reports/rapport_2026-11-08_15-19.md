    document.getElementById('cvFile').addEventListener('change', async (e) => {
      const file = e.target.files[0];
      if (!file) return;
      const arrayBuffer = await file.arrayBuffer();
      const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
      let text = '';
      for (let i = 1; i <= pdf.numPages; i++) {
        const page = await pdf.getPage(i);
        const content = await page.getTextContent();

    function analyzeATS() {
      const cvText = document.getElementById('cvText').value;
      const jobText = document.getElementById('jobText').value;

      if (!cvText.trim() || !jobText.trim()) {
        alert("Veuillez renseigner votre CV et l'offre d'emploi.");
        return;

      const cvWords = new Set(extractKeywords(cvText));

        .sort((a, b) => frequency[b] - frequency[a])

      const missing = [];

      topJobKeywords.forEach(word => {
        if (cvWords.has(word)) matched.push(word);
        else missing.push(word);
      });

      const score = Math.round((matched.length / topJobKeywords.length) * 100);

      // Affichage

${jobText.substring(0, 1500)}
"""

"""`;

      document.getElementById('results').classList.remove('hidden');
