### Le Script Technique Clé en Main (Python + FFmpeg)

Ce script prend une vidéo brute, transcrit la voix, détecte les mots à la milliseconde près, applique un template dynamique (texte jaune/blanc en gras avec contour noir) et exporte la vidéo finale en MP4.

```python
# installation requise (0 €) : pip install faster-whisper
import os
import subprocess
from faster_whisper import WhisperModel

def generate_subtitles(video_path, output_path="output_subtitled.mp4"):
    # 1. Extraction et transcription avec GPU ou CPU (Modèle Whisper Medium)
    print("⏳ Transcription audio en cours via Whisper...")
    model = WhisperModel("base", device="auto", compute_type="auto")
    segments, _ = model.transcribe(video_path, word_timestamps=True)

    # 2. Création du fichier de sous-titres animés (.ass)
    ass_path = "subtitles.ass"
    header = """[Script Info]
ScriptType: v4.00+
PlayResX: 1080
PlayResY: 1920

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Hormozi,Impact,80,&H0000FFFF,&H000000FF,&H00000000,&H80000000,-1,0,0,0,100,100,0,0,1,6,0,10,10,10,960,1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""
    events = []
    for segment in segments:
        for word in segment.words:
            # Format temps ASS : H:MM:SS.cs
            def fmt(t):
                h = int(t // 3600)
                m = int((t % 3600) // 60)
                s = t % 60
                return f"{h}:{m:02d}:{s:05.2f}"

            start = fmt(word.start)
            end = fmt(word.end)
            # Animation mot par mot : majuscules, impact, jaune avec contour noir
            word_clean = word.word.strip().upper()
            events.append(f"Dialogue: 0,{start},{end},Hormozi,,0,0,0,,{word_clean}")

    with open(ass_path, "w", encoding="utf-8") as f:
        f.write(header + "\n".join(events))

    print(" Ingestion des sous-titres dans la vidéo via FFmpeg...")
    # 3. Gravure directe des sous-titres (Hardburn)
    cmd = [
        "ffmpeg", "-y", "-i", video_path,
        "-vf", f"ass={ass_path}",
        "-c:a", "copy",
        "-c:v", "libx264", "-crf", "18", "-preset", "fast",
        output_path
    ]
    subprocess.run(cmd, check=True)
    print(f" Terminé ! Vidéo prête : {output_path}")

# Exemple d'usage :
# generate_subtitles("ma_video_brute.mp4")
```
